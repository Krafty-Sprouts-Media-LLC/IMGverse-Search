# v1.0.36 Orientation and Storage Fixes Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make the Orientation filter actually work for iNaturalist and Openverse, share one set of shape rules between server and browser, and stop blocked session storage from breaking batch clicks. Ship as `v1.0.36`.

**Architecture:** A new browser-safe ES module `app/public/image-shape.js` holds the landscape/portrait/square rules; the server (`providers/index.js`) and the browser (`openverse-client.js`) both import it. The iNaturalist adapter gets a pure, exported mapper that reads `original_dimensions`. Openverse sends its native `aspect_ratio` filter and keeps the shared ratio check as a safety net. A new `app/public/safe-storage.js` wraps every `sessionStorage` call in `app.js`.

**Tech Stack:** Node 20+, Express, ES modules, `node:test` + `node:assert/strict` (run with `npm test` in `app/`), vanilla browser JS. No new dependencies.

**Spec:** [`docs/superpowers/specs/2026-10-14-sise-bulk-keywords-design.md`](../specs/2026-10-14-sise-bulk-keywords-design.md), §7 (bug fixes) and §11 step 1. Everything else in the spec belongs to the later `v2.0.0` plan.

/**
 * Shared image shape / orientation rules.
 *
 * Used by the server (providers/index.js) and the browser
 * (openverse-client.js) so both sides agree on what "landscape",
 * "portrait" and "square" mean. Browser-safe: no Node imports.
 *
 * @package IMGverse-Search
 * @since   1.0.36
 */

'use strict';

export const LANDSCAPE_MIN_RATIO = 1.15;
export const PORTRAIT_MAX_RATIO = 0.87;

const ORIENTATIONS = new Set(['landscape', 'portrait', 'square']);

/**
 * Classify an image by its width / height ratio.
 *
 * @param {number} width
 * @param {number} height
 * @returns {''|'landscape'|'portrait'|'square'} Empty string when size is unknown.
 */
export function shapeOf(width, height) {
  const w = Number(width) || 0;
  const h = Number(height) || 0;
  if (!w || !h) return '';
  const ratio = w / h;
  if (ratio > LANDSCAPE_MIN_RATIO) return 'landscape';
  if (ratio < PORTRAIT_MAX_RATIO) return 'portrait';
  return 'square';
}

/**
 * Whether an image fits the requested orientation.
 * Unknown sizes and unknown orientation values always match.
 *
 * @param {{ width?: number, height?: number }} img
 * @param {string} orientation '' | 'landscape' | 'portrait' | 'square'
 * @returns {boolean}
 */
export function matchesOrientation(img, orientation) {
  if (!ORIENTATIONS.has(orientation)) return true;
  const shape = shapeOf(img?.width, img?.height);
  return !shape || shape === orientation;
}

/**
 * Keep only results that fit the requested orientation.
 *
 * @param {object[]} results
 * @param {string}   orientation
 * @returns {object[]}
 */
export function filterByOrientation(results, orientation) {
  if (!orientation) return results;
  return results.filter((img) => matchesOrientation(img, orientation));
}
```

/**
 * Map iNaturalist observations to canonical ImageResult objects.
 *
 * iNaturalist returns up to `per_page` observations, but each observation can
 * contain many photos. Cap the provider at 20 image results per page so the
 * overall "20 per provider" limit is honoured.
 *
 * @param {object[]} observations - `results` array from the observations API.
 * @returns {object[]} Canonical ImageResult array (max 20).
 */
export function mapObservations(observations) {
  const results = [];

  for (const obs of observations || []) {
    for (const photo of obs.photos || []) {
      const thumbUrl = photo.url?.replace('square', 'large') || photo.url;     // 1024px — right-click Save As gives a usable image
      const fullUrl  = photo.url?.replace('square', 'original') || photo.url;  // original resolution for "Open full image" button

      if (!thumbUrl) continue;

      results.push(
        normalize({
          id:        photo.id,
          provider:  'inaturalist',
          thumbUrl,
          fullUrl,
          // Real size lets the orientation filter work; 0 = unknown (kept by every filter).
          width:     Number(photo.original_dimensions?.width) || 0,
          height:    Number(photo.original_dimensions?.height) || 0,
          alt:       obs.species_guess || obs.taxon?.name || 'Nature photo',
          credit:    obs.user?.login || '',
          creditUrl: obs.user ? `https://www.inaturalist.org/people/${obs.user.login}` : '',
          license:   photo.license_code || 'cc',
          sourceUrl: `https://www.inaturalist.org/observations/${obs.id}`,
        })
      );
    }
  }

  return results.slice(0, 20);
}

// =============================================================================
// src/lib/safe-storage.test.js
// IMGverse Search — Tests for the throw-proof sessionStorage wrapper.
//
// @package IMGverse-Search
// @since   1.0.36
// =============================================================================

'use strict';

import { describe, it } from 'node:test';
import assert from 'node:assert/strict';
import {
  storageGet,
  storageGetJSON,
  storageRemove,
  storageSet,
} from '../../public/safe-storage.js';

function memoryStorage() {
  const map = new Map();
  return {
    getItem: (k) => (map.has(k) ? map.get(k) : null),
    setItem: (k, v) => { map.set(k, String(v)); },
    removeItem: (k) => { map.delete(k); },
  };
}

const throwing = {
  getItem() { throw new Error('SecurityError'); },
  setItem() { throw new Error('QuotaExceededError'); },
  removeItem() { throw new Error('SecurityError'); },
};

describe('safe-storage with working storage', () => {
  it('round-trips values', () => {
    const s = memoryStorage();
    assert.equal(storageSet('a', '1', s), true);
    assert.equal(storageGet('a', s), '1');
    assert.equal(storageRemove('a', s), true);
    assert.equal(storageGet('a', s), null);
  });

  it('parses JSON and falls back on bad JSON', () => {
    const s = memoryStorage();
    s.setItem('ok', '[1,2]');
    s.setItem('bad', '{not json');
    assert.deepEqual(storageGetJSON('ok', [], s), [1, 2]);
    assert.deepEqual(storageGetJSON('bad', [], s), []);
    assert.deepEqual(storageGetJSON('missing', ['x'], s), ['x']);
  });
});

describe('safe-storage with blocked storage', () => {
  it('never throws when every call throws', () => {
    assert.equal(storageGet('a', throwing), null);
    assert.equal(storageSet('a', '1', throwing), false);
    assert.equal(storageRemove('a', throwing), false);
    assert.deepEqual(storageGetJSON('a', [], throwing), []);
  });

  it('never throws when sessionStorage itself is unavailable', () => {
    // Node has no sessionStorage, which models the data: URL case.
    assert.equal(storageGet('a'), null);
    assert.equal(storageSet('a', '1'), false);
    assert.equal(storageRemove('a'), false);
  });
});

 * `sessionStorage` property (private mode, blocked site data, quota
 * exceeded, data: URLs). The UI must keep working without it, so every
 * call here returns a safe value and logs the first failure once.
 *
 * @package IMGverse-Search
 * @since   1.0.36
 */

'use strict';

let warned = false;

function warnOnce(action, key, err) {
  if (warned) return;
  warned = true;
  console.warn(
    `[IMGverse/storage] ${action} "${key}" failed; batch state will not survive a reload:`,
    err?.message || err
  );
}

function resolve(storage) {
  if (storage) return storage;
  try {
    return globalThis.sessionStorage || null;
  } catch {
    return null;
  }
}

/**
 * @param {string} key
 * @param {Storage|null} [storage]
 * @returns {string|null}
 */
export function storageGet(key, storage) {
  const s = resolve(storage);
  if (!s) return null;
  try {
    return s.getItem(key);
  } catch (err) {
    warnOnce('read', key, err);
    return null;
  }
}

/**
 * @param {string} key
 * @param {string} value
 * @param {Storage|null} [storage]
 * @returns {boolean} Whether the value was stored.
 */
export function storageSet(key, value, storage) {
  const s = resolve(storage);
  if (!s) return false;
  try {
    s.setItem(key, value);
    return true;
  } catch (err) {
    warnOnce('write', key, err);
    return false;
  }
}

/**
 * @param {string} key
 * @param {Storage|null} [storage]
 * @returns {boolean} Whether the key was removed.
 */
export function storageRemove(key, storage) {
  const s = resolve(storage);
  if (!s) return false;
  try {
    s.removeItem(key);
    return true;
  } catch (err) {
    warnOnce('remove', key, err);
    return false;
  }
}

/**
 * Read and parse JSON, returning `fallback` when missing, blocked or invalid.
 *
 * @param {string} key
 * @param {*} fallback
 * @param {Storage|null} [storage]
 * @returns {*}
 */
export function storageGetJSON(key, fallback, storage) {
  const raw = storageGet(key, storage);
  if (raw == null) return fallback;
  try {
    return JSON.parse(raw);
  } catch {
    return fallback;
  }
}
```
